package main

  

import "fmt"

  

func binSearch(numbers []int, need int) {

    var s, e int = 0, len(numbers) - 1

  

    for s <= e {

        var mid int = (s + e) / 2

        if need > numbers[mid] {

            s = mid + 1

            continue

        }

  

        if need < numbers[mid] {

            e = mid - 1

            continue

        }

  

        if need == numbers[mid] {

            fmt.Println("число есть")

            return

        }

    }

  

    fmt.Println("числа нет")

}

  

func main() {

    var numbers []int

    var need int = 100

  

    for number := range 101 {

        if number == 0 {

            continue

        }

  

        numbers = append(numbers, number)

    }

  

    binSearch(numbers, need)

}
