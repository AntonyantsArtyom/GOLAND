package main

import (

    "fmt"

    "math"

)

  

func findLowerNode(costs map[string]int, processed map[string]bool) string {

    lowerCost := math.MaxInt32

    lowerCostNode := ""

  

    for node, cost := range costs {

        if cost < lowerCost && !processed[node] {

            lowerCost = cost

            lowerCostNode = node

        }

    }

  

    return lowerCostNode

}

  

func dijkstra(graph map[string]map[string]int, costs map[string]int, parents map[string]string, processed map[string]bool) {

    node := findLowerNode(costs, processed)

  

    for node != "" {

        cost := costs[node]

        neighbours := graph[node]

  

        for neighbour, weight := range neighbours {

            newCost := cost + weight

  

            if costs[neighbour] > newCost {

                costs[neighbour] = newCost

                parents[neighbour] = node

            }

        }

  

        processed[node] = true

  

        node = findLowerNode(costs, processed)

  

    }

  

    fmt.Printf("Минимальная стоимость до финиша: %d\n", costs["finish"])

    fmt.Println("Родительские связи для восстановления пути:", parents)

}

  

func main() {

    graph := make(map[string]map[string]int)

  

    graph["start"] = map[string]int{"A": 10}

    graph["A"] = map[string]int{"B": 10, "C": 20}

    graph["B"] = map[string]int{"finish": 10}

    graph["D"] = map[string]int{"finish": 1}

  

    costs := map[string]int{

        "A":      10,

        "B":      math.MaxInt32,

        "C":      math.MaxInt32,

        "D":      math.MaxInt32,

        "finish": math.MaxInt32,

    }

  

    parents := map[string]string{

        "A":      "start",

        "B":      "",

        "C":      "",

        "D":      "",

        "finish": "",

    }

  

    processed := make(map[string]bool)

  

    dijkstra(graph, costs, parents, processed)

}
