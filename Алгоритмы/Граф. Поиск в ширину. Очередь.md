package main

  

import "fmt"

  

func isFly(s string) bool {

    return s == "попугай"

}

  

func main() {

    friends := make(map[string][]string)

  

    friends["я"] = []string{"биба", "боба"}

    friends["боба"] = []string{"лаврентий палыч", "барсик"}

    friends["барсик"] = []string{"муська", "тыква"}

    friends["тыква"] = []string{"муська", "попугай"}

  

    queue := append([]string{}, friends["я"]...)

    searched := make(map[string]bool)

  

    for len(queue) > 0 {

        person := queue[0]

        queue = queue[1:]

  

        if searched[person] == true {

            continue

        }

  

        searched[person] = true

  

        if isFly(person) {

            fmt.Println("у вас есть летающий знакомый")

            return

        }

  

        queue = append(queue, friends[person]...)

  

    }

  

    fmt.Println("Летающих знакомых нет")

}
