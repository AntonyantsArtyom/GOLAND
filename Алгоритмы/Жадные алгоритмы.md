```go
package main

  
import "fmt"
  

func main() {

    change := 48

    coins := []int{10, 5, 3, 1}

  

    fmt.Println("выдача сдачи")

  

    for _, coin := range coins {

        for change-coin >= 0 {

            change -= coin

            fmt.Println("выдали " + fmt.Sprint(coin) + " рублей, " + fmt.Sprint(change) + " рублей осталось")

        }

    }

}
```
