```go
package main

  

import "fmt"

  

func quick(n []int) []int {

    if len(n) <= 1 {

        return n

    }

  

    p := n[0]

  

    l := []int{}

    r := []int{}

  

    for _, e := range n[1:] {

        if e < p {

            l = append(l, e)

        } else {

            r = append(r, e)

        }

    }

  

    return append(append(quick(l), p), quick(r)...)

}

  

func main() {

    fmt.Println("Отсортированный массив:", quick([]int{150, 2, 10, 300, 10, 20, 30}))

}
```
